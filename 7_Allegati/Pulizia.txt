Per disabilitare:
# 1. Disabilita Hyper-V
Disable-WindowsOptionalFeature -Online -FeatureName Microsoft-Hyper-V-All -NoRestart

# 2. Disabilita Windows Sandbox
Disable-WindowsOptionalFeature -Online -FeatureName Containers-DisposableClientVM -NoRestart

# 3. Disabilita la piattaforma macchina virtuale
Disable-WindowsOptionalFeature -Online -FeatureName VirtualMachinePlatform -NoRestart

# 4. Disabilita Windows Hypervisor Platform
Disable-WindowsOptionalFeature -Online -FeatureName HypervisorPlatform -NoRestart

# 5. Disabilita Windows Subsystem for Linux (WSL)
Disable-WindowsOptionalFeature -Online -FeatureName Microsoft-Windows-Subsystem-Linux -NoRestart

# 6. Disabilita il lancio dell'hypervisor all'avvio
bcdedit /set hypervisorlaunchtype off
